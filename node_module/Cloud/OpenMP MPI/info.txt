#include <stdio.h> 
#include <omp.h>

int main() {
    int sum = 0;
    int n = 10; // Sum numbers from 1 to 10

    #pragma omp parallel for reduction(+:sum)
    for (int i = 1; i <= n; i++) {
        sum += i;
    }

    printf("Total Sum: %d\n", sum);
    return 0;
}


#include <stdio.h>
#include <omp.h>

int main() {
    int n = 5; // Calculate 5! (5 * 4 * 3 * 2 * 1)
    long long fact = 1; // Use long long as factorials grow rapidly

    #pragma omp parallel for reduction(*:fact)
    for (int i = 1; i <= n; i++) {
        fact *= i;
    }

    printf("Factorial of %d: %lld\n", n, fact);
    return 0;
}

#include <stdio.h>
#include <omp.h>

int main() {
    int a[] = {10, 50, 20, 90, 30, 70, 40};
    int n = 7;
    int max1 = -1, max2 = -1; // Assuming positive numbers

    // Pass 1: Find the absolute maximum
    #pragma omp parallel for reduction(max:max1)
    for (int i = 0; i < n; i++) {
        if (a[i] > max1) max1 = a[i];
    }

    // Pass 2: Find the maximum that is strictly less than max1
    #pragma omp parallel for reduction(max:max2)
    for (int i = 0; i < n; i++) {
        if (a[i] > max2 && a[i] < max1) max2 = a[i];
    }

    printf("First Max: %d\n", max1);
    printf("Second Max: %d\n", max2);

    return 0;
}

#include <stdio.h>
#include <omp.h>

int main() {
    int n = 5;
    int a[] = {1, 2, 3, 4, 5};
    int b[] = {10, 20, 30, 40, 50};
    int result[5];

    // Parallelize the loop so different threads handle different indices
    #pragma omp parallel for
    for (int i = 0; i < n; i++) {
        result[i] = a[i] + b[i];
    }

    // Print the final result array
    printf("Result Array: \n");
    for (int i = 0; i < n; i++) {
        printf("%d + %d = %d\n", a[i], b[i], result[i]);
    }

    return 0;
}

from mpi4py import MPI

comm = MPI.COMM_WORLD
rank = comm.Get_rank()

# Assign a value based on the process rank (e.g., 10, 20, 30...)
my_val = (rank + 1) * 10

# Add all values together
total_sum = comm.reduce(my_val, op=MPI.SUM, root=0)

if rank == 0:
    print(f"Total Sum: {total_sum}")

my_val = (rank + 1) * 5

# Find the highest value among all processes
max_val = comm.reduce(my_val, op=MPI.MAX, root=0)

if rank == 0:

from mpi4py import MPI

comm = MPI.COMM_WORLD
rank = comm.Get_rank()

# Assign a value based on rank
my_val = (rank + 1) * 5

# Gather all values from all processes into a list on root (Process 0)
all_vals = comm.gather(my_val, root=0)

if rank == 0:
    # set() removes duplicates, sorted() orders them, [-2] gets the second to last
    second_max = sorted(set(all_vals))[-2]
    print(f"Second Maximum Value: {second_max}")

from mpi4py import MPI

comm = MPI.COMM_WORLD
rank = comm.Get_rank()
size = comm.Get_size()

# Each process represents one part of the factorial sequence
my_val = rank + 1

# Multiply all values together
factorial = comm.reduce(my_val, op=MPI.PROD, root=0)

if rank == 0:
    print(f"Factorial of {size}: {factorial}")
